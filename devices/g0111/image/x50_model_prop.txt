# board,name} and ro.build.product (Lumen keeps its own) and persist.sys.* (the GSI's namespace). compose_x50.sh checks on the build server that each line below is a line

# no other line of the file passes the rule. persist.vendor.xgimi.hd_server=2 starts the vendor human detection
